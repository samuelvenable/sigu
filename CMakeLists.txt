cmake_minimum_required(VERSION 3.10)
set(CMAKE_CXX_STANDARD 17)
project(sigu)
if(UNIX AND NOT APPLE)
	set(SDL_X11_XSCRNSAVER OFF)
	set(SDL_X11_XINPUT OFF)
	set(SDL_X11_XFIXES OFF)
	set(SDL_X11_XTEST OFF)
endif()
set(IFD_USE_LIBRARY_BUILD ON)
set(IFD_USE_BASIC_ICONS ON)
add_subdirectory(libfiledialogs)
add_compile_definitions(CREATE_CONTEXT)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
else()
	if(WIN32)
		set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -ggdb -static")
	elseif(APPLE)
		set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -ggdb -ObjC++")
		if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
			set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc -static-libstdc++")
		endif()
	else()
		if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
			if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc -static-libstdc++")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc -static-libstdc++")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "DragonFly")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc")
			endif()
		endif()
		set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -ggdb")
	endif()
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_TOOLCHAIN_FILE)
	include("${CMAKE_TOOLCHAIN_FILE}")
endif(CMAKE_TOOLCHAIN_FILE)
if(UNIX)
	find_package(PkgConfig REQUIRED)
endif()
find_package(OpenGL REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/cl-bootstrap.cmake")
cl_repo()
if(CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
	set(CATALOG_REQ_TYPE PREFER_SHARED)
else()
	set(CATALOG_REQ_TYPE PREFER_STATIC)
endif()
if(APPLE)
	set(SOURCES sigu.cpp system.cpp shmem.mm)
else()
	set(SOURCES sigu.cpp system.cpp)
endif()
add_executable(sigu ${SOURCES})
cl_add_dep(sigu SDL3)
if(CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
	target_include_directories(sigu PRIVATE ${LIBUSB_LIBRARIES} "${CMAKE_CURRENT_SOURCE_DIR}/libfiledialogs/source")
else()
	target_include_directories(sigu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/libfiledialogs/source")
endif()
target_link_libraries(sigu PRIVATE ${OPENGL_LIBRARIES})
if(WIN32)
	target_link_libraries(sigu PRIVATE ws2_32 dxgi psapi)
elseif(UNIX AND NOT APPLE)
	if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD" OR CMAKE_SYSTEM_NAME STREQUAL "DragonFly" OR CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
		target_link_libraries(sigu PRIVATE ${CMAKE_DL_LIBS} X11 kvm pthread)
	elseif(CMAKE_SYSTEM_NAME STREQUAL "NetBSD" OR CMAKE_SYSTEM_NAME STREQUAL "SunOS")
		pkg_search_module(HWLOC REQUIRED hwloc)
		include_directories(${HWLOC_INCLUDE_DIRS})
		link_directories(${HWLOC_LIBRARY_DIRS})
		target_link_libraries(sigu PRIVATE ${CMAKE_DL_LIBS} ${HWLOC_LIBRARIES} X11 pthread)
	else()
		target_link_libraries(sigu PRIVATE ${CMAKE_DL_LIBS} X11 pthread)
	endif()
elseif(APPLE)
	target_link_libraries(sigu PRIVATE "-framework Cocoa -framework CoreGraphics -framework Metal -framework Foundation" ${CMAKE_DL_LIBS})
endif()
