cmake_minimum_required(VERSION 3.10)
set(CMAKE_CXX_STANDARD 20)
project(sigu)
if(APPLE)
	set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
	set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64")
endif()
if((WIN32 OR APPLE) OR (UNIX AND NOT APPLE AND IFD_USE_CATALOG_PACKAGES))
	set(CATALOG_LOCAL_CACHE ON)
	set(CATALOG_FORCE_source_SDL3 TRUE)
endif()
if(UNIX AND NOT APPLE AND IFD_USE_CATALOG_PACKAGES)
	set(SDL_X11_XSCRNSAVER OFF)
	set(SDL_X11_XINPUT OFF)
	set(SDL_X11_XFIXES OFF)
	set(SDL_X11_XTEST OFF)
endif()
if((WIN32 OR APPLE) OR (UNIX AND NOT APPLE AND IFD_USE_CATALOG_PACKAGES))
	set(BUILD_SHARED_LIBS OFF)
	set(IFD_USE_LIBRARY_BUILD ON)
	set(IFD_USE_BASIC_ICONS ON)
endif()
add_compile_definitions(CREATE_CONTEXT)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
else()
	if(WIN32)
		set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -ggdb -static")
	elseif(APPLE)
		set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -ggdb -ObjC++")
		if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
			set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc -static-libstdc++")
		endif()
	else()
		if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
			if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc -static-libstdc++")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc -static-libstdc++")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "DragonFly")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc")
			elseif(CMAKE_SYSTEM_NAME STREQUAL "SunOS")
				set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libgcc")
			endif()
		endif()
		set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -g -ggdb")
	endif()
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_TOOLCHAIN_FILE)
	include("${CMAKE_TOOLCHAIN_FILE}")
endif(CMAKE_TOOLCHAIN_FILE)
if(UNIX)
	find_package(PkgConfig REQUIRED)
endif()
if(UNIX AND NOT APPLE)
	find_package(OpenGL REQUIRED)
endif()
if((WIN32 OR APPLE) OR (UNIX AND NOT APPLE AND IFD_USE_CATALOG_PACKAGES))
	include("${CMAKE_CURRENT_SOURCE_DIR}/cl-bootstrap.cmake")
	cl_repo()
	if(CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
		set(CATALOG_REQ_TYPE PREFER_SHARED)
	else()
		set(CATALOG_REQ_TYPE PREFER_STATIC)
	endif()
endif()
if(APPLE)
	set(SOURCES sigu.cpp system.cpp shmem.mm)
else()
	set(SOURCES sigu.cpp system.cpp)
endif()
add_executable(sigu ${SOURCES})
if((WIN32 OR APPLE) OR (UNIX AND NOT APPLE AND IFD_USE_CATALOG_PACKAGES))
	cl_add_dep(sigu SDL3)
endif()
set_property(TARGET sigu PROPERTY POSITION_INDEPENDENT_CODE ON)
set_target_properties(sigu PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES OUTPUT_NAME "sigu" PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
if((WIN32 OR APPLE) OR (UNIX AND NOT APPLE AND IFD_USE_CATALOG_PACKAGES))
	add_subdirectory(libfiledialogs)
	if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
		add_custom_command(TARGET sigu PRE_LINK COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/PRE_LINK.bat")
		target_link_directories(sigu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
	endif()
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD" OR CMAKE_SYSTEM_NAME STREQUAL "DragonFly")
	target_include_directories(sigu PUBLIC "/usr/local/include")
	target_link_directories(sigu PUBLIC "/usr/local/lib")
elseif(CMAKE_SYSTEM_NAME STREQUAL "NetBSD")
	target_include_directories(sigu PUBLIC "/usr/local/include" "/usr/pkg/include" "/usr/X11R7/include")
	target_link_directories(sigu PUBLIC "/usr/local/lib" "/usr/pkg/lib" "/usr/X11R7/lib")
elseif(CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
	target_include_directories(sigu PUBLIC "/usr/local/include" "/usr/X11R6/include")
	target_link_directories(sigu PUBLIC ${LIBUSB_LIBRARIES} "/usr/local/lib" "/usr/X11R6/lib")
endif()
if((WIN32 OR APPLE) OR (UNIX AND NOT APPLE AND IFD_USE_CATALOG_PACKAGES))
	target_include_directories(sigu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/libfiledialogs/source")
endif()
if(UNIX AND NOT APPLE)
	target_link_libraries(sigu PUBLIC ${OPENGL_LIBRARIES})
endif()
if(WIN32)
	target_link_libraries(sigu PUBLIC ws2_32 dxgi psapi)
elseif(UNIX AND NOT APPLE)
	if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD" OR CMAKE_SYSTEM_NAME STREQUAL "DragonFly" OR CMAKE_SYSTEM_NAME STREQUAL "OpenBSD")
		target_link_libraries(sigu PUBLIC ${CMAKE_DL_LIBS} X11 kvm pthread)
	elseif(CMAKE_SYSTEM_NAME STREQUAL "NetBSD" OR CMAKE_SYSTEM_NAME STREQUAL "SunOS")
		pkg_search_module(HWLOC REQUIRED hwloc)
		include_directories(${HWLOC_INCLUDE_DIRS})
		link_directories(${HWLOC_LIBRARY_DIRS})
		target_link_libraries(sigu PUBLIC ${CMAKE_DL_LIBS} ${HWLOC_LIBRARIES} X11 pthread)
	else()
		target_link_libraries(sigu PUBLIC ${CMAKE_DL_LIBS} X11 pthread)
	endif()
elseif(APPLE)
	target_link_libraries(sigu PUBLIC "-framework Cocoa -framework CoreGraphics -framework Metal -framework Foundation" ${CMAKE_DL_LIBS})
endif()
target_link_libraries(sigu PUBLIC filedialogs)
